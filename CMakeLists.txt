cmake_minimum_required(VERSION 3.16)
project(IMOL VERSION 10.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(config ${CMAKE_CONFIGURATION_TYPES})
    string(TOUPPER "${config}" config_upper)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_BINARY_DIR}/bin")
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_BINARY_DIR}/bin")
    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_BINARY_DIR}/lib")
endforeach()
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Network Widgets)

file(GLOB IMOL_CORE_HEADERS CONFIGURE_DEPENDS imol/include/core/*.h)
file(GLOB IMOL_CORE_SOURCES CONFIGURE_DEPENDS imol/src/core/*.cpp)
add_library(imolCore SHARED ${IMOL_CORE_HEADERS} ${IMOL_CORE_SOURCES} imol/src/core/asset.qrc)
target_compile_definitions(imolCore PRIVATE CORE_LIBRARY)
target_include_directories(imolCore PUBLIC imol/include PRIVATE imol/include/core)
target_link_libraries(imolCore PUBLIC Qt5::Core Qt5::Network)
add_library(imol::core ALIAS imolCore)

file(GLOB_RECURSE IMOL_BWAF_HEADERS CONFIGURE_DEPENDS imol/include/bwaf/*.h)
file(GLOB_RECURSE IMOL_BWAF_SOURCES CONFIGURE_DEPENDS imol/src/bwaf/*.cpp imol/src/bwaf/*.ui)
add_library(imolBwaf SHARED ${IMOL_BWAF_HEADERS} ${IMOL_BWAF_SOURCES})
target_compile_definitions(imolBwaf PRIVATE BWAF_LIBRARY MODULE_NAME="imol.ui")
target_include_directories(imolBwaf PRIVATE imol/include/bwaf)
target_link_libraries(imolBwaf PUBLIC imolCore Qt5::Widgets)
add_library(imol::widgets ALIAS imolBwaf)

file(GLOB_RECURSE IMOLX_HEADERS CONFIGURE_DEPENDS imolx/include/*.h imolx/src/*.h)
file(GLOB_RECURSE IMOLX_SOURCES CONFIGURE_DEPENDS imolx/src/*.cpp imolx/src/*.ui)
add_library(imolx SHARED ${IMOLX_HEADERS} ${IMOLX_SOURCES} imolx/imolx_asset.qrc)
target_compile_definitions(imolx PRIVATE IMOLX_LIBRARY MODULE_NAME="imolx")
target_include_directories(imolx PUBLIC imolx/include PRIVATE imolx imolx/include/imol imolx/dependencies/include)
target_link_libraries(imolx PUBLIC imolBwaf)

set(HOTKEY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/imol/src/program/util/QHotkey/QHotkey")
set(HOTKEY_SOURCES ${HOTKEY_DIR}/qhotkey.cpp ${HOTKEY_DIR}/qhotkey.h ${HOTKEY_DIR}/qhotkey_p.h)
if(WIN32)
    list(APPEND HOTKEY_SOURCES ${HOTKEY_DIR}/qhotkey_win.cpp)
elseif(APPLE)
    enable_language(OBJCXX)
    list(APPEND HOTKEY_SOURCES ${HOTKEY_DIR}/qhotkey_mac.cpp)
else()
    find_package(Qt5 REQUIRED COMPONENTS X11Extras)
    find_package(X11 REQUIRED)
    list(APPEND HOTKEY_SOURCES ${HOTKEY_DIR}/qhotkey_x11.cpp)
endif()
file(GLOB PROGRAM_SOURCES CONFIGURE_DEPENDS imol/src/program/*.cpp imol/src/program/*.h imol/src/program/*.ui)
qt5_add_big_resources(PROGRAM_RESOURCES imol/src/program/imol_asset.qrc)
add_executable(imolProgram ${PROGRAM_SOURCES} ${PROGRAM_RESOURCES} ${HOTKEY_SOURCES})
target_include_directories(imolProgram PRIVATE ${HOTKEY_DIR})
target_link_libraries(imolProgram PRIVATE imolBwaf)
if(WIN32 AND MSVC)
    target_sources(imolProgram PRIVATE
        imol/src/program/util/StackWalker/BaseException.cpp
        imol/src/program/util/StackWalker/StackWalker.cpp)
elseif(APPLE)
    target_link_libraries(imolProgram PRIVATE "-framework Carbon" "-framework AppKit")
elseif(UNIX)
    target_link_libraries(imolProgram PRIVATE Qt5::X11Extras X11::X11)
endif()
add_dependencies(imolProgram imolx)

foreach(target imolCore imolBwaf imolx imolProgram)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /bigobj)
    endif()
endforeach()

set(IMOLX_LIBRARY_FILE "${CMAKE_SHARED_LIBRARY_PREFIX}imolx${CMAKE_SHARED_LIBRARY_SUFFIX}")
file(MAKE_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
configure_file(runtime/internal_license.json.in "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/internal_license.json" @ONLY)
configure_file(runtime/config.ini "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/config.ini" COPYONLY)

option(IMOL_BUILD_TESTS "Build the xservice wire-contract client" OFF)
if(IMOL_BUILD_TESTS)
    add_executable(imol_xservice_client tests/xservice_client.cpp)
    target_link_libraries(imol_xservice_client PRIVATE imolx)
endif()
