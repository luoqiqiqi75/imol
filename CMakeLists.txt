cmake_minimum_required(VERSION 3.16)
project(IMOL VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

option(BUILD_SHARED_LIBS "Build shared IMOL libraries" ON)
option(IMOL_BUILD_EXAMPLES "Build the IMOL data-tree example" ON)

find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Network)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Network)

file(GLOB IMOL_HEADERS CONFIGURE_DEPENDS include/imol/*.h)
file(GLOB IMOL_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(imol ${IMOL_HEADERS} ${IMOL_SOURCES} src/asset.qrc)
add_library(imol::core ALIAS imol)
target_compile_definitions(imol PRIVATE CORE_LIBRARY)
if(NOT BUILD_SHARED_LIBS)
    target_compile_definitions(imol PUBLIC IMOL_STATIC_LIBRARY)
endif()
target_include_directories(imol PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
)
target_include_directories(imol PRIVATE include/imol)
target_link_libraries(imol PUBLIC Qt${QT_VERSION_MAJOR}::Core Qt${QT_VERSION_MAJOR}::Network)
if(MSVC)
    target_compile_options(imol PRIVATE /utf-8)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/widgets")
    option(IMOL_BUILD_WIDGETS "Build the IMOL widget framework" OFF)
    if(IMOL_BUILD_WIDGETS)
        find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
        set(CMAKE_AUTOUIC ON)
        file(GLOB_RECURSE IMOL_WIDGET_HEADERS CONFIGURE_DEPENDS include/bwaf/*.h)
        file(GLOB_RECURSE IMOL_WIDGET_SOURCES CONFIGURE_DEPENDS widgets/*.cpp widgets/*.ui)
        add_library(imol_widgets ${IMOL_WIDGET_HEADERS} ${IMOL_WIDGET_SOURCES})
        add_library(imol::widgets ALIAS imol_widgets)
        target_compile_definitions(imol_widgets PRIVATE BWAF_LIBRARY)
        target_include_directories(imol_widgets PRIVATE include/bwaf)
        target_link_libraries(imol_widgets PUBLIC imol::core Qt${QT_VERSION_MAJOR}::Widgets)
        if(MSVC)
            target_compile_options(imol_widgets PRIVATE /utf-8)
        endif()
    endif()
endif()

if(IMOL_BUILD_EXAMPLES)
    add_executable(imol_example examples/basic.cpp)
    target_link_libraries(imol_example PRIVATE imol::core)
endif()
